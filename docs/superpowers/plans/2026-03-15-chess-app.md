# Chess App (v1) Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a cross-platform Flutter chess app with offline AI play (Stockfish) and Lichess puzzle challenges for iOS and Android.

**Architecture:** Clean Architecture with three layers per feature (presentation / domain / data). Domain layer is pure Dart with no external dependencies, enabling full unit-test coverage of game logic. Stockfish runs on-device via FFI; puzzles are served from a bundled SQLite database with FTS5 for theme filtering.

**Tech Stack:** Flutter, Dart, flutter_riverpod (state), go_router (navigation), chess (rules engine), stockfish_flutter (AI FFI), sqflite (puzzle DB), shared_preferences (persistence), flutter_svg (SVG pieces), mocktail (test mocking)

  ```dart
    );
    final epMoves = epChess.moves({'square': 'e5', 'verbose': true}) as List;

    print('En passant detected: $hasEP');

    final staleChess = Chess.fromFEN('k7/8/1R6/8/8/8/8/1R5K b - - 0 1');
  ```

  ```dart
  import 'dart:async';
  import 'package:stockfish_flutter/stockfish_flutter.dart';

  Future<void> main() async {
    final stockfish = Stockfish();
    final completer = Completer<String>();

    stockfish.stdout.listen((line) {
      print('Stockfish: $line');
      if (line.startsWith('bestmove')) {
        completer.complete(line);
      }
    });

    // Wait for engine ready
    await Future.delayed(const Duration(seconds: 1));
    stockfish.stdin = 'uci';
    await Future.delayed(const Duration(milliseconds: 500));
    stockfish.stdin = 'isready';
    await Future.delayed(const Duration(milliseconds: 500));

    // Request a move from starting position, depth 5
    stockfish.stdin = 'setoption name Skill Level value 10';
    stockfish.stdin = 'position startpos';
    stockfish.stdin = 'go depth 5';

    final result = await completer.future.timeout(const Duration(seconds: 15));
    print('Got bestmove line: $result');
    final move = result.split(' ')[1];
    assert(move.length == 4 || move.length == 5, 'UCI move format wrong: $move');

    stockfish.dispose();
    print('stockfish_flutter spike: PASSED');
  }
  ```

  ```dart
  import 'app_colors.dart';

  class BoardTheme {

    final String name;
    final String key;

    const BoardTheme({
      required this.lightSquare,
      required this.darkSquare,
      required this.name,
      required this.key,
    });

    static const greenClean = BoardTheme(
      lightSquare: AppColors.greenLightSquare,
      darkSquare: AppColors.greenDarkSquare,
      name: 'Green & Clean',
      key: 'greenClean',
    );

    static const classicWood = BoardTheme(
      lightSquare: AppColors.woodLightSquare,
      darkSquare: AppColors.woodDarkSquare,
      name: 'Classic Wood',
      key: 'classicWood',
    );

    static const List<BoardTheme> all = [greenClean, classicWood];

    static BoardTheme fromKey(String key) =>
        all.firstWhere((t) => t.key == key, orElse: () => greenClean);

    @override
    bool operator ==(Object other) => other is BoardTheme && other.key == key;

    @override
    int get hashCode => key.hashCode;
  }
  ```

  ```dart
  import 'chess_engine.dart';
  import 'game_repository.dart';
  import 'game_state.dart';
  import 'models.dart';

  // These providers are overridden in main.dart with real implementations.
  final gameRepositoryProvider = Provider<GameRepository>((ref) {
    throw UnimplementedError('gameRepositoryProvider not overridden');
  });

  final chessEngineProvider = Provider<ChessEngine>((ref) {
    throw UnimplementedError('chessEngineProvider not overridden');
  });

  final gameNotifierProvider =
      StateNotifierProvider<GameNotifier, GameState?>(GameNotifier.new);

  class GameNotifier extends StateNotifier<GameState?> {
    final Ref _ref;

    GameNotifier(this._ref) : super(null);

    GameRepository get _repo => _ref.read(gameRepositoryProvider);
    ChessEngine get _engine => _ref.read(chessEngineProvider);

    void startGame({
      required Side playerColor,
      required DifficultyLevel difficulty,
    }) {
      _repo.reset();
      final result = _repo.loadPosition(GameState.kStartFen);
      state = GameState(
        fen: result.fen,
        history: const [],
        legalMoves: result.legalMoves,
        playerColor: playerColor,
        difficulty: difficulty,
        status: GameStatus.playing,
      );
    }

    Future<void> applyPlayerMove(String uciMove) async {
      final current = state;
      if (current == null || !current.isPlayerTurn) return;

      // Apply player's move
      final moveResult = _repo.applyMove(uciMove);
      final playerMove = Move(uci: uciMove, san: moveResult.sanMove);
      final afterPlayer = current.copyWith(
        fen: moveResult.fen,
        history: [...current.history, playerMove],
        legalMoves: moveResult.legalMoves,
        status: moveResult.status,
        isAiThinking: moveResult.status == GameStatus.playing,
      );
      state = afterPlayer;

      if (moveResult.status != GameStatus.playing) return;

      // Request AI move
      try {
        final bestUci = await _engine.getBestMove(
          moveResult.fen,
          current.difficulty,
        );
        final aiResult = _repo.applyMove(bestUci);
        final aiMove = Move(uci: bestUci, san: aiResult.sanMove);
        state = afterPlayer.copyWith(
          fen: aiResult.fen,
          history: [...afterPlayer.history, aiMove],
          legalMoves: aiResult.legalMoves,
          status: aiResult.status,
          isAiThinking: false,
        );
      } catch (_) {
        // Engine error handled by UI via isAiThinking remaining true
        state = afterPlayer.copyWith(isAiThinking: false);
        rethrow;
      }
    }

    void resign() {
      state = state?.copyWith(status: GameStatus.checkmate);
    }

    void clearGame() {
      state = null;
    }
  }
  ```

  ```dart
  import 'package:chess/chess.dart' as ch;
  import '../domain/game_repository.dart';
  import '../domain/game_state.dart';
  import '../domain/models.dart';

  class ChessRepositoryImpl implements GameRepository {
    ch.Chess _chess = ch.Chess();

    @override
    GamePositionResult loadPosition(String fen) {
      _chess = ch.Chess.fromFEN(fen);
      return GamePositionResult(
        fen: _chess.fen,
        legalMoves: _legalMovesUci(),
      );
    }

    @override
    GameMoveResult applyMove(String uciMove) {
      final from = uciMove.substring(0, 2);
      final to = uciMove.substring(2, 4);
      final promotion = uciMove.length == 5 ? uciMove[4] : null;

      // Get SAN from verbose move list before applying
      final verboseMoves = _chess.moves({'verbose': true}) as List;
      final match = verboseMoves.cast<Map>().firstWhere(
        (m) =>
            m['from'] == from &&
            m['to'] == to &&
            (promotion == null || m['promotion'] == promotion),
        orElse: () => {},
      );
      final san = match['san'] as String? ?? uciMove;

      _chess.move({
        'from': from,
        'to': to,
        if (promotion != null) 'promotion': promotion,
      });

      return GameMoveResult(
        fen: _chess.fen,
        legalMoves: _legalMovesUci(),
        status: _status(),
        sanMove: san,
      );
    }

    @override
    void reset() => _chess = ch.Chess();

    List<String> _legalMovesUci() {
      final verboseMoves = _chess.moves({'verbose': true}) as List;
      return verboseMoves.cast<Map>().map((m) {
        final from = m['from'] as String;
        final to = m['to'] as String;
        final promo = m['promotion'] as String?;
        return promo != null ? '$from$to$promo' : '$from$to';
      }).toList();
    }

    GameStatus _status() {
      if (_chess.in_checkmate) return GameStatus.checkmate;
      if (_chess.in_stalemate) return GameStatus.stalemate;
      if (_chess.in_draw) return GameStatus.draw;
      return GameStatus.playing;
    }
  }
  ```

  ```dart
  import 'dart:async';
  import 'package:stockfish_flutter/stockfish_flutter.dart'; // verify package name from spike
  import '../domain/chess_engine.dart';
  import '../domain/models.dart';

  class StockfishService implements ChessEngine {
    Stockfish? _stockfish;
    Completer<String>? _moveCompleter;
    StreamSubscription<String>? _sub;
    bool _ready = false;

    /// Call once at app start. Returns false if initialization fails.
    Future<bool> initialize() async {
      try {
        _stockfish = Stockfish();
        final readyCompleter = Completer<void>();

        _sub = _stockfish!.stdout.listen((line) {
          if (line == 'uciok' || line == 'readyok') {
            if (!readyCompleter.isCompleted) readyCompleter.complete();
          }
          if (line.startsWith('bestmove')) {
            final parts = line.split(' ');
            if (parts.length >= 2) {
              _moveCompleter?.complete(parts[1]);
              _moveCompleter = null;
            }
          }
        });

        _stockfish!.stdin = 'uci';
        await readyCompleter.future.timeout(const Duration(seconds: 5));
        _stockfish!.stdin = 'isready';
        await Future.delayed(const Duration(milliseconds: 300));
        _ready = true;
        return true;
      } catch (_) {
        _ready = false;
        return false;
      }
    }

    bool get isReady => _ready;

    @override
    Future<String> getBestMove(String fen, DifficultyLevel difficulty) async {
      if (!_ready || _stockfish == null) {
        throw StateError('Stockfish not initialized');
      }
      _moveCompleter = Completer<String>();
      _stockfish!.stdin = 'setoption name Skill Level value ${difficulty.skillLevel}';
      _stockfish!.stdin = 'position fen $fen';
      _stockfish!.stdin = 'go depth ${difficulty.searchDepth}';

      return _moveCompleter!.future.timeout(
        const Duration(seconds: 10),
        onTimeout: () {
          _stockfish!.stdin = 'stop';
          _moveCompleter = null;
          throw TimeoutException('Stockfish timed out after 10s');
        },
      );
    }

    @override
    void dispose() {
      _sub?.cancel();
      _stockfish?.dispose();
      _stockfish = null;
      _ready = false;
    }
  }
  ```

  ```dart
  import 'dart:convert';
  import 'package:shared_preferences/shared_preferences.dart';
  import '../domain/game_state.dart';
  import '../domain/models.dart';

  class GamePersistenceService {
    static const _kFen = 'game.fen';
    static const _kHistory = 'game.history';
    static const _kPlayerColor = 'game.playerColor';
    static const _kDifficulty = 'game.difficulty';
    static const _kStatus = 'game.status';

    Future<void> saveGame(GameState state) async {
      final prefs = await SharedPreferences.getInstance();
      await prefs.setString(_kFen, state.fen);
      await prefs.setString(
        _kHistory,
        jsonEncode(state.history.map((m) => m.uci).toList()),
      );
      await prefs.setString(_kPlayerColor, state.playerColor.name);
      await prefs.setString(_kDifficulty, state.difficulty.name);
      await prefs.setString(_kStatus, state.status.name);
    }

    /// Returns null if no saved game, or if saved data is corrupt.
    Future<GameState?> restoreGame() async {
      try {
        final prefs = await SharedPreferences.getInstance();
        final statusStr = prefs.getString(_kStatus);
        if (statusStr != 'playing') return null;

        final fen = prefs.getString(_kFen);
        final historyJson = prefs.getString(_kHistory);
        final colorStr = prefs.getString(_kPlayerColor);
        final diffStr = prefs.getString(_kDifficulty);

        if (fen == null || historyJson == null || colorStr == null || diffStr == null) {
          return null;
        }

        final historyUci = (jsonDecode(historyJson) as List).cast<String>();
        final playerColor =
            Side.values.firstWhere((s) => s.name == colorStr);
        final difficulty = DifficultyLevel.fromName(diffStr);

        return GameState(
          fen: fen,
          // SANs are not stored — UI shows history from domain history list.
          // Restore with empty SAN; full history with SAN is rebuilt via
          // GameRepository.applyMove replay if needed for display.
          history: historyUci.map((u) => Move(uci: u, san: '')).toList(),
          legalMoves: const [], // recomputed on game load
          playerColor: playerColor,
          difficulty: difficulty,
          status: GameStatus.playing,
        );
      } catch (_) {
        await clearGame();
        return null;
      }
    }

    Future<void> clearGame() async {
      final prefs = await SharedPreferences.getInstance();
      await prefs.remove(_kFen);
      await prefs.remove(_kHistory);
      await prefs.remove(_kPlayerColor);
      await prefs.remove(_kDifficulty);
      await prefs.remove(_kStatus);
    }
  }
  ```

  ```dart
  class BoardWidget extends StatefulWidget {
    final String fen;
    final List<String> legalMoves; // UCI strings
    final BoardTheme boardTheme;
    final String pieceSet;
    final bool showCoordinates;
    final bool flipped;
    final bool interactive;
    final void Function(String uciMove) onMove;
    final String? lastMoveFrom;
    final String? lastMoveTo;

    const BoardWidget({
      super.key,
      required this.fen,
      required this.legalMoves,
      required this.boardTheme,
      required this.pieceSet,
      required this.showCoordinates,
      required this.flipped,
      required this.interactive,
      required this.onMove,
      this.lastMoveFrom,
      this.lastMoveTo,
    });

    @override
    State<BoardWidget> createState() => _BoardWidgetState();
  }

  class _BoardWidgetState extends State<BoardWidget> {
    String? _selectedSquare;

    List<String> get _legalDestsForSelected {
      if (_selectedSquare == null) return [];
      return widget.legalMoves
          .where((m) => m.substring(0, 2) == _selectedSquare)
          .map((m) => m.substring(2, 4))
          .toList();
    }

    void _onSquareTapped(String square) {
      if (!widget.interactive) return;

      if (_selectedSquare == null) {
        // Select if a piece of the correct color is here
        if (_hasPieceToMove(square)) {
          setState(() => _selectedSquare = square);
        }
        return;
      }

      // Try to move
      final dest = square;
      final from = _selectedSquare!;
      final matchingMoves = widget.legalMoves
          .where((m) => m.substring(0, 2) == from && m.substring(2, 4) == dest)
          .toList();

      if (matchingMoves.isEmpty) {
        // Re-select if tapping another own piece
        if (_hasPieceToMove(square)) {
          setState(() => _selectedSquare = square);
        } else {
          setState(() => _selectedSquare = null);
        }
        return;
      }

      // Check for promotion
      final isPromotion = matchingMoves.any((m) => m.length == 5);
      if (isPromotion) {
        _showPromotionModal(from, dest, matchingMoves);
        return;
      }

      setState(() => _selectedSquare = null);
      widget.onMove(matchingMoves.first);
    }

    bool _hasPieceToMove(String square) {
      // Determine active color from FEN
      final activeColor = widget.fen.split(' ')[1]; // 'w' or 'b'
      // Parse FEN to get piece at square
      final chess = ch.Chess.fromFEN(widget.fen);
      final piece = chess.get(square);
      if (piece == null) return false;
      final pieceColor = piece['color'] as String;
      return pieceColor == activeColor;
    }

    void _showPromotionModal(String from, String to, List<String> moves) {
      showModalBottomSheet<String>(
        context: context,
        builder: (_) => _PromotionPicker(
          pieceSet: widget.pieceSet,
          activeColor: widget.fen.split(' ')[1],
        ),
      ).then((piece) {
        if (piece == null) {
          setState(() => _selectedSquare = null);
          return;
        }
        setState(() => _selectedSquare = null);
        widget.onMove('$from$to$piece');
      });
    }

    @override
    Widget build(BuildContext context) {
      return LayoutBuilder(
        builder: (context, constraints) {
          final size = constraints.maxWidth;
          final squareSize = size / 8;

          return SizedBox(
            width: size,
            height: size,
            child: Stack(
              children: [
                // Base board squares
                _buildSquares(squareSize),
                // Highlights
                HighlightLayer(
                  selectedSquare: _selectedSquare,
                  legalMoveDests: _legalDestsForSelected,
                  lastMoveFrom: widget.lastMoveFrom,
                  lastMoveTo: widget.lastMoveTo,
                  squareSize: squareSize,
                  flipped: widget.flipped,
                ),
                // Pieces
                _buildPieces(squareSize),
                // Coordinates
                if (widget.showCoordinates)
                  CoordinateLabels(
                    squareSize: squareSize,
                    flipped: widget.flipped,
                  ),
              ],
            ),
          );
        },
      );
    }

    Widget _buildSquares(double squareSize) {
      return GridView.builder(
        physics: const NeverScrollableScrollPhysics(),
        gridDelegate: const SliverGridDelegateWithFixedCrossAxisCount(
          crossAxisCount: 8,
        ),
        itemCount: 64,
        itemBuilder: (_, index) {
          final (col, row) = _indexToColRow(index);
          final square = _colRowToSquare(col, row);
          final isLight = (col + row) % 2 == 0;
          return SquareWidget(
            isLight: isLight,
            boardTheme: widget.boardTheme,
            onTap: () => _onSquareTapped(square),
          );
        },
      );
    }

    Widget _buildPieces(double squareSize) {
      final chess = ch.Chess.fromFEN(widget.fen);
      return Stack(
        children: [
          for (final square in _allSquares())
            Builder(builder: (_) {
              final piece = chess.get(square);
              if (piece == null) return const SizedBox.shrink();
              final color = piece['color'] as String; // 'w' or 'b'
              final type = piece['type'] as String;   // 'p', 'r', etc.
              final assetName = '${color}${type.toUpperCase()}';
              final (col, row) = _squareToDisplayColRow(square);
              return AnimatedPositioned(
                duration: const Duration(milliseconds: 200),
                curve: Curves.easeInOut,
                left: col * squareSize,
                top: row * squareSize,
                child: PieceWidget(
                  piece: assetName,
                  pieceSet: widget.pieceSet,
                  size: squareSize,
                ),
              );
            }),
        ],
      );
    }

    (int, int) _indexToColRow(int index) {
      final col = index % 8;
      final row = index ~/ 8;
      return (col, row);
    }

    (int, int) _squareToDisplayColRow(String square) {
      final col = square.codeUnitAt(0) - 'a'.codeUnitAt(0);
      final row = 8 - int.parse(square[1]);
      return widget.flipped ? (7 - col, 7 - row) : (col, row);

      for (final file in 'abcdefgh'.split('')) {

      }
      return squares;
    }
  ```

  ```dart
  import 'dart:io';
  import 'package:path/path.dart' as p;
  import 'package:sqflite_common_ffi/sqflite_ffi.dart';

  // Run: dart run tools/build_puzzles_db.dart <path-to-csv> [limit]
  // Example: dart run tools/build_puzzles_db.dart lichess_db_puzzle.csv 50000
  Future<void> main(List<String> args) async {
    if (args.isEmpty) {
      print('Usage: dart run tools/build_puzzles_db.dart <csv_path> [limit]');
      exit(1);
    }

    final csvPath = args[0];
    final limit = args.length > 1 ? int.parse(args[1]) : 50000;
    final outputPath = p.join(Directory.current.path, 'assets', 'puzzles.db');

    // Delete existing DB
    final outputFile = File(outputPath);
    if (outputFile.existsSync()) outputFile.deleteSync();

    sqfliteFfiInit();
    final db = await databaseFactoryFfi.openDatabase(outputPath);

    // Create schema
    await db.execute('''
      CREATE TABLE puzzles (
        id          TEXT PRIMARY KEY,
        fen         TEXT NOT NULL,
        moves       TEXT NOT NULL,
        rating      INTEGER,
        themes      TEXT,
        popularity  INTEGER,
        solved      INTEGER DEFAULT 0
      )
    ''');
    await db.execute('CREATE INDEX idx_rating     ON puzzles(rating)');
    await db.execute('CREATE INDEX idx_popularity ON puzzles(popularity DESC)');
    await db.execute('CREATE INDEX idx_solved     ON puzzles(solved)');
    await db.execute('''
      CREATE VIRTUAL TABLE puzzles_fts USING fts5(
        id UNINDEXED,
        themes,
        content=puzzles,
        content_rowid=rowid
      )
    ''');

    // Import CSV
    final lines = File(csvPath).readAsLinesSync();
    final header = lines.first.split(',');
    final idIdx = header.indexOf('PuzzleId');
    final fenIdx = header.indexOf('FEN');
    final movesIdx = header.indexOf('Moves');
    final ratingIdx = header.indexOf('Rating');
    final popularityIdx = header.indexOf('Popularity');
    final themesIdx = header.indexOf('Themes');

    var count = 0;
    await db.transaction((txn) async {
      final batch = txn.batch();
      for (final line in lines.skip(1)) {
        if (count >= limit) break;
        final cols = line.split(',');
        if (cols.length < 8) continue;
        batch.insert('puzzles', {
          'id': cols[idIdx],
          'fen': cols[fenIdx],
          'moves': cols[movesIdx],
          'rating': int.tryParse(cols[ratingIdx]) ?? 1500,
          'themes': cols[themesIdx],
          'popularity': int.tryParse(cols[popularityIdx]) ?? 0,
        });
        count++;
      }
      await batch.commit(noResult: true);
    });

    // Rebuild FTS5 index
    await db.execute("INSERT INTO puzzles_fts(puzzles_fts) VALUES('rebuild')");

    await db.close();
    print('Built puzzles.db with $count puzzles → $outputPath');
  }
  ```

  ```dart
  import 'dart:io';
  import 'dart:async';
  import 'package:flutter/services.dart';
  import 'package:path/path.dart' as p;
  import 'package:path_provider/path_provider.dart';
  import 'package:sqflite/sqflite.dart';

  class PuzzleDatabase {
    static Database? _db;

    static Future<Database?> open() async {
      if (_db != null) return _db;
      final docsDir = await getApplicationDocumentsDirectory();
      final dbPath = p.join(docsDir.path, 'puzzles.db');

      if (!File(dbPath).existsSync()) {
        await _copyFromAssets(dbPath);
      }

      try {
        _db = await openDatabase(dbPath, readOnly: false);
        return _db;
      } catch (_) {
        return null; // Corrupt DB — puzzles section disabled
      }
    }

    static Future<void> _copyFromAssets(String destPath) async {
      const timeoutDuration = Duration(seconds: 30);
      try {
        await Future.any([
          _doCopy(destPath),
          Future.delayed(timeoutDuration).then(
            (_) => throw TimeoutException('DB copy timed out'),
          ),
        ]);
      } catch (_) {
        // Delete partial file on failure
        final f = File(destPath);
        if (f.existsSync()) f.deleteSync();
        rethrow;
      }
    }

    static Future<void> _doCopy(String destPath) async {
      final data = await rootBundle.load('assets/puzzles.db');
      final bytes = data.buffer.asUint8List();
      await File(destPath).writeAsBytes(bytes, flush: true);
    }

    static Future<int?> countPuzzles() async {
      final db = await open();
      if (db == null) return null;
      final result = await db.rawQuery('SELECT COUNT(*) as c FROM puzzles');
      return (result.first['c'] as int?);
    }
  ```

  ```dart
        FROM puzzles p
        ${filter.theme != null ? 'JOIN puzzles_fts f ON p.rowid = f.rowid' : ''}
        ${filter.theme != null ? "WHERE puzzles_fts MATCH ? AND p.rating BETWEEN ? AND ?" : whereClause}
      ''', [
        if (filter.theme != null) ...[filter.theme!, filter.minRating, filter.maxRating],
        if (filter.theme == null) ...whereArgs,
      ]);
      return (rows.first['c'] as int?) ?? 0;
    }

    @override
    Future<Puzzle?> getPuzzleById(String id) async {
      final rows = await _db.query('puzzles', where: 'id = ?', whereArgs: [id]);
      return rows.isEmpty ? null : Puzzle.fromMap(rows.first);
    }

    @override
    Future<Puzzle?> getDailyPuzzle() async {
      final prefs = await SharedPreferences.getInstance();
      final count = prefs.getInt('puzzle.count') ?? await _cacheCount(prefs);
      if (count == 0) return null;

      final now = DateTime.now().toUtc();
      final index = (now.year * 10000 + now.month * 100 + now.day) % count;

      final rows = await _db.rawQuery(
        'SELECT * FROM puzzles ORDER BY rowid LIMIT 1 OFFSET ?',
        [index],
      );
      return rows.isEmpty ? null : Puzzle.fromMap(rows.first);
    }

    @override
    Future<void> markSolved(String puzzleId) async {
      await _db.update(
        'puzzles',
        {'solved': 1},
        where: 'id = ?',
        whereArgs: [puzzleId],
      );
    }

    @override
    Future<Puzzle?> getNextPuzzle(String currentId, PuzzleFilter filter) async {
      // Get the current puzzle's popularity to find the next one
      final current = await getPuzzleById(currentId);
      if (current == null) return null;

      final (whereClause, whereArgs) = _buildWhere(filter);
      final rows = await _db.rawQuery('''
        SELECT p.*
        FROM puzzles p
        ${filter.theme != null ? 'JOIN puzzles_fts f ON p.rowid = f.rowid' : ''}
        ${filter.theme != null
            ? "WHERE puzzles_fts MATCH ? AND p.rating BETWEEN ? AND ? AND p.popularity < ?"
            : '$whereClause AND p.popularity < ?'}
        ORDER BY p.popularity DESC
        LIMIT 1
      ''', [
        if (filter.theme != null)
          ...[filter.theme!, filter.minRating, filter.maxRating, current.popularity]
        else
          ...[...whereArgs, current.popularity],
      ]);
      return rows.isEmpty ? null : Puzzle.fromMap(rows.first);
    }

    (String, List<dynamic>) _buildWhere(PuzzleFilter filter) {
      final where = 'WHERE p.rating BETWEEN ? AND ?';
      final args = <dynamic>[filter.minRating, filter.maxRating];
      return (where, args);
    }

    Future<int> _cacheCount(SharedPreferences prefs) async {
      final count = await _db.rawQuery('SELECT COUNT(*) as c FROM puzzles');
      final n = (count.first['c'] as int?) ?? 0;
      await prefs.setInt('puzzle.count', n);
      return n;
    }
  }
  ```

  ```dart
  import 'package:flutter_riverpod/flutter_riverpod.dart';
  import 'package:chess_app/features/game/data/chess_repository_impl.dart';
  import 'puzzle.dart';
  import 'puzzle_filter.dart';
  import 'puzzle_repository.dart';
  import 'puzzle_session.dart';

  final puzzleRepositoryProvider = Provider<PuzzleRepository>((ref) {
    throw UnimplementedError('puzzleRepositoryProvider not overridden');
  });

  final puzzleFilterProvider = StateProvider<PuzzleFilter>((ref) => PuzzleFilter.empty);

  final puzzleNotifierProvider =
      StateNotifierProvider<PuzzleNotifier, PuzzleSession?>(PuzzleNotifier.new);

  class PuzzleNotifier extends StateNotifier<PuzzleSession?> {
    final Ref _ref;
    final _chessRepo = ChessRepositoryImpl();

    PuzzleNotifier(this._ref) : super(null);

    PuzzleRepository get _repo => _ref.read(puzzleRepositoryProvider);

    Future<void> loadPuzzle(Puzzle puzzle) async {
      // Load FEN from before the setup move
      _chessRepo.loadPosition(puzzle.fen);

      // Apply setup move (moves[0]) automatically
      final setupMove = puzzle.moves[0];
      final afterSetup = _chessRepo.applyMove(setupMove);

      state = PuzzleSession(
        puzzle: puzzle,
        fen: afterSetup.fen,
        nextMoveIndex: 1, // first user move
        hintsUsed: 0,
        status: PuzzleSessionStatus.playing,
      );
    }

    Future<bool> submitMove(String uciMove) async {
      final session = state;
      if (session == null || session.isComplete) return false;

      final expected = session.expectedMove;
      if (expected == null) return false;

      // Normalize comparison (strip promotion char case differences)
      if (uciMove.toLowerCase() != expected.toLowerCase()) {
        // Wrong move — do not advance, keep state for retry
        return false;
      }

      // Correct — apply user move
      _chessRepo.loadPosition(session.fen);
      final afterUser = _chessRepo.applyMove(uciMove);
      var nextIndex = session.nextMoveIndex + 1;
      var currentFen = afterUser.fen;

      // Check if puzzle complete
      if (nextIndex >= session.puzzle.moves.length) {
        state = session.copyWith(
          fen: currentFen,
          nextMoveIndex: nextIndex,
          status: PuzzleSessionStatus.solved,
        );
        await _repo.markSolved(session.puzzle.id);
        return true;
      }

      // Apply opponent response automatically
      _chessRepo.loadPosition(currentFen);
      final opponentMove = session.puzzle.moves[nextIndex];
      final afterOpponent = _chessRepo.applyMove(opponentMove);
      nextIndex++;
      currentFen = afterOpponent.fen;

      // Check if puzzle complete after opponent response
      if (nextIndex >= session.puzzle.moves.length) {
        state = session.copyWith(
          fen: currentFen,
          nextMoveIndex: nextIndex,
          status: PuzzleSessionStatus.solved,
        );
        await _repo.markSolved(session.puzzle.id);
      } else {
        state = session.copyWith(
          fen: currentFen,
          nextMoveIndex: nextIndex,
        );
      }
      return true;
    }

    void useHint() {
      final session = state;
      if (session == null || session.hintsUsed >= 2) return;
      state = session.copyWith(hintsUsed: session.hintsUsed + 1);
    }

    void clear() => state = null;
  }
  ```

  ```dart
            Padding(
  ```

  ```dart
                  fontWeight: FontWeight.w600,
                  color: session.isComplete ? AppColors.successGreen : AppColors.textPrimary,

              ),
            ),

                aspectRatio: 1,

                    // Success overlay
                    if (session.isComplete)
                      _SuccessOverlay(
                        puzzle: puzzle,
                        onNext: () async {
                          final filter = ref.read(puzzleFilterProvider);
                          final repo = ref.read(puzzleRepositoryProvider);
                          final next = await repo.getNextPuzzle(puzzle.id, filter);
                          if (next != null && context.mounted) {
                            ref.read(puzzleNotifierProvider.notifier).loadPuzzle(next);
                            context.pushReplacement('/puzzles/${next.id}');
                          }
                        },

              Padding(
                padding: const EdgeInsets.all(12),

                      .map((t) => Chip(label: Text(t, style: const TextStyle(fontSize: 12))))
                      .toList(),
                ),
              ),
          ],
        ),
      );
    }

    Widget _buildBoard(
      BuildContext context,
      WidgetRef ref,
      PuzzleSession session,
      dynamic settings,
      bool flipped,
    ) {
      final hint1Square = session.hintsUsed >= 1 ? session.expectedMove?.substring(0, 2) : null;
      final hint2Square = session.hintsUsed >= 2 ? session.expectedMove?.substring(2, 4) : null;

      // Build legalMoves list: only show hint destination if hint2 used
      final boardLegalMoves = hint1Square != null && hint2Square != null
          ? ['$hint1Square$hint2Square'] // show single dot on dest
          : <String>[];

      return BoardWidget(
        fen: session.fen,
        legalMoves: session.isComplete ? [] : boardLegalMoves,
        boardTheme: settings.boardTheme,
        pieceSet: settings.pieceSet,
        showCoordinates: settings.showCoordinates,
        flipped: flipped,
        interactive: !session.isComplete,
        onMove: (uciMove) async {
          final correct = await ref
              .read(puzzleNotifierProvider.notifier)
              .submitMove(uciMove);
          if (!correct && context.mounted) {
            ScaffoldMessenger.of(context).showSnackBar(
              const SnackBar(
                content: Text('Not the best move — try again'),
                duration: Duration(seconds: 1),
              ),
            );
          }
        },
        lastMoveFrom: null,
        lastMoveTo: null,
      );
    }
  }
  ```

  ```dart
  class SettingsNotifier extends StateNotifier<AppSettings> {
    SettingsNotifier() : super(const AppSettings()) {
      _load();
    }

    static const _kBoardTheme = 'settings.boardTheme';
    static const _kPieceSet = 'settings.pieceSet';
    static const _kSound = 'settings.sound';
    static const _kLegalHints = 'settings.legalHints';
    static const _kCoordinates = 'settings.coordinates';

    Future<void> _load() async {
      final prefs = await SharedPreferences.getInstance();
      state = AppSettings(
        boardTheme: BoardTheme.fromKey(prefs.getString(_kBoardTheme) ?? 'greenClean'),
        pieceSet: prefs.getString(_kPieceSet) ?? 'cburnett',
        soundEnabled: prefs.getBool(_kSound) ?? true,
        showLegalHints: prefs.getBool(_kLegalHints) ?? true,
        showCoordinates: prefs.getBool(_kCoordinates) ?? true,
      );
    }

    Future<void> update(AppSettings settings) async {
      state = settings;
      final prefs = await SharedPreferences.getInstance();
      await prefs.setString(_kBoardTheme, settings.boardTheme.key);
      await prefs.setString(_kPieceSet, settings.pieceSet);
      await prefs.setBool(_kSound, settings.soundEnabled);
      await prefs.setBool(_kLegalHints, settings.showLegalHints);
      await prefs.setBool(_kCoordinates, settings.showCoordinates);
    }
  }
  ```
